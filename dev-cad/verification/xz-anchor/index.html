Anchors on XZ-profile extrusion

What this example makes

Left and right markers attach to their respective front-top corners with mirrored spatial results.

Forge code

This is the complete Forge source used for this example.

class Board:
    width = 40.0
    height = 20.0
    depth = 12.0

    def build(self):
        half_width = half(self.width)
        with Sketch(name="Profile", plane="XZ") as sketch:
            Polygon2D(
                [
                    ("lower_left", (-half_width, 0)),
                    ("lower_right", (half_width, 0)),
                    ("upper_right", (half_width, self.height)),
                    ("upper_left", (-half_width, self.height)),
                ],
                plane="XZ",
                world=True,
            )
        return sketch.extrude(self.depth).with_material(
            "test board",
            density_g_cm3=0.55,
        )


class RightMarker:
    def build(self):
        return Box(10, 10, 10)


class LeftMarker:
    def build(self):
        return Box(10, 10, 10)


class XzAnchor:
    board_width = 40.0
    board_height = 20.0

    def build(self):
        board = Board(
            width=self.board_width,
            height=self.board_height,
        )
        right_marker = RightMarker().attach(
            board,
            parent=Anchor.FRONT_TOP_RIGHT,
            child=Anchor.BACK_TOP_RIGHT,
        )
        left_marker = LeftMarker().attach(
            board,
            parent=Anchor.FRONT_TOP_LEFT,
            child=Anchor.BACK_TOP_LEFT,
        )
        return board + right_marker + left_marker

API used

model = cq.Assembly()

model.add(marker, name="right_marker", loc=cq.Location((15, -5, 15)))
model.add(marker, name="left_marker", loc=cq.Location((-15, -5, 15)))

"""build123d reference: XZ-oriented board and two front corner markers."""
from build123d import Align, Box, BuildPart, Locations

bottom = (Align.CENTER, Align.CENTER, Align.MIN)

with BuildPart() as model:
    with Locations((0, 6, 0)):
        Box(40, 12, 20, align=bottom)

    with Locations((15, -5, 15), (-15, -5, 15)):
        Box(10, 10, 10)
