Face anchor follows parent geometry

What this example makes

The attached block stays on the parent's right face.

Forge code

This is the complete Forge source used for this example.

class Base:
    width = 40.0

    def build(self):
        return Box(self.width, 30, 10, anchor=Anchor.BOTTOM)


class Block:
    def build(self):
        return Box(20, 20, 10, anchor=Anchor.BOTTOM)


class ParametricAnchor:
    base_width = 40.0

    def build(self):
        base = Base(width=self.base_width)
        block = Block().attach(
            base,
            parent=Anchor.RIGHT,
            child=Anchor.LEFT,
        )
        return base + block

"""CadQuery reference for a block placed against the base's right face."""

import cadquery as cq


base_width = 40.0
child_width = 20.0

block = cq.Solid.makeBox(child_width, 20, 10, cq.Vector(-child_width / 2, -10, 0))

model = cq.Assembly()
model.add(base, name="base")

    block,
    name="block",
    loc=cq.Location((base_width / 2 + child_width / 2, 0, 0)),

"""build123d reference: centered block positioned against a right face."""
from build123d import Align, Box, BuildPart, Locations

base_width, child_width = 40.0, 20.0
bottom = (Align.CENTER, Align.CENTER, Align.MIN)

with BuildPart() as model:
    Box(base_width, 30, 10, align=bottom)
    with Locations((base_width / 2 + child_width / 2, 0, 0)):
        Box(child_width, 20, 10, align=bottom)
