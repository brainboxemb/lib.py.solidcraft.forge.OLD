Rounded Sketch pad and standalone world-plane cut

What this example makes

The result is one joined body: an 80 x 60 x 5 mm base plate with a centered 40 x 30 x 8 mm rounded pad on top. The Rounded pad sketch has one closed profile made from four straight edges and four corner arcs. SolidCraft 0.42 also adds one construction diagonal between opposite arc centres; that construction line is not part of the profile.

Forge code

This is the complete Forge source used for this example.

"""Qualification: attached rounded pad plus standalone world-plane cut."""


class RoundedRectangleSketch:
    base_width = 80.0
    base_depth = 60.0
    base_height = 5.0
    pad_width = 40.0
    pad_depth = 30.0
    radius = 5.0
    pad_height = 8.0
    cut_width = 10.0
    cut_height = 2.0
    cut_y = 22.0
    cut_z = 2.5
    cut_length = 100.0

    def build(self):
        body = Box(
            self.base_width,
            self.base_depth,
            self.base_height,
            anchor=Anchor.BOTTOM,
        )

        with Sketch(name="Rounded pad", on=body) as pad:
            pad.add(
                RoundedRectangle2D(
                    self.pad_width,
                    self.pad_depth,
                    self.radius,
                ),
            )

        body = body + pad.extrude(self.pad_height)

        with Sketch(name="World cut", plane="YZ") as cut:
            cut.add(
                Rectangle2D(self.cut_width, self.cut_height),
                offset=(self.cut_y, self.cut_z),
            )

        return body - cut.extrude(self.cut_length, sym=True)

"""CadQuery reference for rounded pad plus independent world-plane tunnel."""

import cadquery as cq


base_width = 80.0
base_depth = 60.0
base_height = 5.0
pad_width = 40.0
pad_depth = 30.0
radius = 5.0
pad_height = 8.0
cut_width = 10.0
cut_height = 2.0
cut_y = 22.0
cut_z = 2.5
cut_length = 100.0

base = cq.Solid.makeBox(
    base_width,
    base_depth,
    base_height,
    cq.Vector(-base_width / 2, -base_depth / 2, 0),
)
pad = (
    cq.Workplane("XY")
    .rect(pad_width - 2 * radius, pad_depth - 2 * radius)
    .offset2D(radius, kind="arc")
    .extrude(pad_height)
    .translate((0, 0, base_height))
    .val()
)
body = base.fuse(pad)
cut = cq.Solid.makeBox(
    cut_length,
    cut_width,
    cut_height,
    cq.Vector(-cut_length / 2, cut_y - cut_width / 2, cut_z - cut_height / 2),
)
result = body.cut(cut)

model = cq.Assembly()
model.add(result, name="rounded-pad-world-cut")

"""build123d reference: rounded pad plus independent world-plane tunnel."""
from build123d import (
    Align,
    Axis,
    Box,
    BuildPart,
    BuildSketch,
    Locations,
    Mode,
    RectangleRounded,
    extrude,
)

base_width, base_depth, base_height = 80.0, 60.0, 5.0
pad_width, pad_depth, radius, pad_height = 40.0, 30.0, 5.0, 8.0
cut_width, cut_height, cut_y, cut_z, cut_length = 10.0, 2.0, 22.0, 2.5, 100.0
bottom = (Align.CENTER, Align.CENTER, Align.MIN)
centered = (Align.CENTER, Align.CENTER, Align.CENTER)

with BuildPart() as model:
    Box(base_width, base_depth, base_height, align=bottom)

    with BuildSketch(model.faces().sort_by(Axis.Z)[-1]):
        RectangleRounded(pad_width, pad_depth, radius)

    extrude(amount=pad_height)

    with Locations((0, cut_y, cut_z)):
        Box(
            cut_length,
            cut_width,
            cut_height,
            align=centered,
            mode=Mode.SUBTRACT,
        )
